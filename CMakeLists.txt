cmake_minimum_required(VERSION 3.16)
project(pi_agent CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pi_core INTERFACE)
target_include_directories(pi_core INTERFACE src vendor)
find_package(Threads REQUIRED)
target_link_libraries(pi_core INTERFACE Threads::Threads)
option(PI_WITH_TLS "Enable HTTPS via OpenSSL" OFF)
if(PI_WITH_TLS)
  find_package(OpenSSL REQUIRED)
  target_compile_definitions(pi_core INTERFACE CPPHTTPLIB_OPENSSL_SUPPORT)
  target_link_libraries(pi_core INTERFACE OpenSSL::SSL OpenSSL::Crypto)
endif()
add_executable(pi src/main.cpp)
target_link_libraries(pi PRIVATE pi_core)
enable_testing()
add_executable(test_session tests/test_session.cpp)
target_link_libraries(test_session PRIVATE pi_core)
add_test(NAME session COMMAND test_session)
add_executable(test_tools tests/test_tools.cpp)
target_link_libraries(test_tools PRIVATE pi_core)
add_test(NAME tools COMMAND test_tools)
add_executable(test_config tests/test_config.cpp)
target_link_libraries(test_config PRIVATE pi_core)
add_test(NAME config COMMAND test_config)
